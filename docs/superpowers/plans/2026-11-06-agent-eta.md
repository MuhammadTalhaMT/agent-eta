# agent-eta Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a public Claude Code plugin that estimates build time realistically for agentic work (agent time vs user time, range, risk, speed-ups) and learns from the user's own transcripts.

**Architecture:** Claude only decomposes and tags tasks; dependency-free Node scripts do all maths. `estimate.mjs` turns tags into minutes, `speedups.mjs` re-runs the estimate with changes to price each tip, `measure.mjs` lazily finds finished estimates, measures real time from Claude Code transcripts and updates a per-user `calibration.json`. Three skills (`eta`, `eta-done`, `eta-stats`) call one CLI.

**Tech Stack:** Node.js ≥ 18 ESM (`.mjs`), no npm dependencies, `node:test` (Node ≥ 22 for test globs), Claude Code plugin format (skills).

**Spec:** `docs/superpowers/specs/2026-11-06-agent-eta-design.md`

**Deviations from the spec (decided while planning):**
1. Commands are implemented as **skills** (`skills/eta`, `skills/eta-done`, `skills/eta-stats`), invoked as `/agent-eta:eta` etc. A skill is told its own base directory when it loads, so it can locate `scripts/cli.mjs` reliably; plain command files can't.
2. Records store the **project path**, not the session id. Measurement uses the project's transcript folder within the time window, which also covers work that spans several sessions.
3. `AGENT_ETA_HOME` (data dir) and `AGENT_ETA_PROJECTS` (transcripts root) env vars override paths — used by tests and the consistency check.
4. The consistency check is a script (`tools/consistency.mjs`) that drives `claude -p` 3 times, instead of a `claude plugin eval` suite.

**Verified transcript facts (checked on this machine, Claude Code 2.1.319):**
- Transcripts: `~/.claude/projects/<encoded path>/<session>.jsonl`; encoded path = project path with every non-alphanumeric char replaced by `-` (`C:\Users\muham` → `C--Users-muham`, `D:\Payday 3` → `D--Payday-3`).
- Subagents: `<session>/subagents/agent-<id>.jsonl` + `agent-<id>.meta.json` with a `description` field.
- Lines have `type` (`user`, `assistant`, plus many non-message types) and ISO `timestamp`. Lines are **not** always in time order — sort.
- Real human prompts: `type: "user"`, no `isMeta`, no `isCompactSummary`, no `toolUseResult`, content is a string or has `text` blocks, and text does not start with `<task-notification>`, `<bash-stdout>`, `<bash-stderr>`, `<local-command-stdout>`, `<local-command-stderr>` or `[Request interrupted`.

import { readFileSync, mkdirSync } from 'node:fs';
import { homedir } from 'node:os';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const HERE = dirname(fileURLToPath(import.meta.url));
export const DEFAULTS_PATH = join(HERE, '..', '..', 'data', 'defaults.json');

export function dataDir(env = process.env) {
  const dir = env.AGENT_ETA_HOME || env.CLAUDE_PLUGIN_DATA || join(homedir(), '.claude', 'agent-eta');
  mkdirSync(dir, { recursive: true });
  return dir;
}

export function loadDefaults(path = DEFAULTS_PATH) {
  return JSON.parse(readFileSync(path, 'utf8'));
}

export function emptyCalibration() {
  return { confirmed: 0, cells: {}, workflow: {}, replyDelay: null };
}

export function loadCalibration(dir) {
  let text;
  try {
    text = readFileSync(join(dir, 'calibration.json'), 'utf8');
  } catch {
    return { calibration: emptyCalibration(), warning: null };
  }
  try {
    return { calibration: { ...emptyCalibration(), ...JSON.parse(text) }, warning: null };
  } catch {
    return { calibration: emptyCalibration(), warning: 'calibration.json is corrupt; using default numbers' };
  }
}

export function effectiveParams(defaults, calibration) {
  const p = structuredClone(defaults);
  if (calibration.confirmed < defaults.minConfirmed) return p;
  for (const [kind, sizes] of Object.entries(p.base)) {
    for (const size of Object.keys(sizes)) {
      sizes[size] *= calibration.cells[`${kind}.${size}`] ?? 1;
    }
  }
  for (const w of Object.keys(p.workflowFactor)) {
    p.workflowFactor[w] *= calibration.workflow[w] ?? 1;
  }
  if (calibration.replyDelay != null) p.replyDelay = calibration.replyDelay;
  return p;
}

export function calibrationStatus(defaults, calibration) {
  const left = defaults.minConfirmed - calibration.confirmed;
  if (left > 0) {
    return `default numbers (${left} more confirmed estimate${left === 1 ? '' : 's'} until personal)`;
  }
  return `personal numbers (from ${calibration.confirmed} confirmed estimates)`;
}
```

```js
const SIZES = ['S', 'M', 'L'];
const KINDS = ['mechanical', 'integration', 'judgment'];
const FLAGS = ['manual_test', 'external_service', 'new_dependency'];
const WORKFLOWS = ['inline', 'subagent-driven'];
const PHASES = ['brainstorm', 'spec', 'plan'];

function fail(field, why) {
  throw new Error(`Invalid tags: ${field} ${why}`);
}

const oneOf = list => `must be one of ${list.join(', ')}`;

export function validateInput(input) {
  if (!input || typeof input !== 'object') fail('input', 'must be an object');
  if (!WORKFLOWS.includes(input.workflow)) fail('workflow', oneOf(WORKFLOWS));

  const phases = input.phases ?? [];
  if (!Array.isArray(phases)) fail('phases', 'must be an array');
  phases.forEach((p, i) => { if (!PHASES.includes(p)) fail(`phases[${i}]`, oneOf(PHASES)); });

  if (!Array.isArray(input.tasks) || input.tasks.length === 0) fail('tasks', 'must be a non-empty array');

  const ids = new Set();
  input.tasks.forEach((t, i) => {
    const at = `tasks[${i}]`;
    if (!t || typeof t !== 'object') fail(at, 'must be an object');
    if (!Number.isInteger(t.id)) fail(`${at}.id`, 'must be an integer');
    if (ids.has(t.id)) fail(`${at}.id`, `duplicates id ${t.id}`);
    ids.add(t.id);
    if (typeof t.title !== 'string' || !t.title.trim()) fail(`${at}.title`, 'must be a non-empty string');
    if (!SIZES.includes(t.size)) fail(`${at}.size`, oneOf(SIZES));
    if (!KINDS.includes(t.kind)) fail(`${at}.kind`, oneOf(KINDS));
    for (const key of ['flags', 'files', 'depends_on']) {
      if (t[key] !== undefined && !Array.isArray(t[key])) fail(`${at}.${key}`, 'must be an array');
    }
    (t.flags ?? []).forEach((f, j) => { if (!FLAGS.includes(f)) fail(`${at}.flags[${j}]`, oneOf(FLAGS)); });
  });

  input.tasks.forEach((t, i) => {
    (t.depends_on ?? []).forEach((d, j) => {
      if (!ids.has(d)) fail(`tasks[${i}].depends_on[${j}]`, `references unknown task ${d}`);
    });
  });

  return {
    workflow: input.workflow,
    phases: [...phases],
    tasks: input.tasks.map(t => ({
      id: t.id,
      title: t.title,
      size: t.size,
      kind: t.kind,
      flags: [...(t.flags ?? [])],
      files: [...(t.files ?? [])],
      depends_on: [...(t.depends_on ?? [])],
    })),
  };
}
```

import { validateInput } from './lib/validate.mjs';

export function estimate(rawInput, params, options = {}) {
  const input = validateInput(rawInput);
  const { frontLoad = false, batchTests = false, parallel = null } = options;
  const wf = params.workflowFactor[input.workflow];

  const tasks = input.tasks.map(t => {
    const spread = params.spread[t.kind] * (t.flags.includes('external_service') ? params.externalSpread : 1);
    const agent = params.base[t.kind][t.size] * wf + (t.flags.includes('new_dependency') ? params.depMinutes : 0);
    const manualTest = t.flags.includes('manual_test');
    return {
      id: t.id,
      title: t.title,
      kind: t.kind,
      size: t.size,
      agent,
      manualTest,
      userActive: manualTest ? params.manualTestActive[t.size] : 0,
      extra: agent * (spread - 1),
    };
  });

  let phaseAgent = 0;
  let phaseActive = 0;
  for (const ph of input.phases) {
    phaseAgent += params.phaseAgent[ph];
    phaseActive += ph === 'brainstorm'
      ? params.phaseActive.brainstormPerQuestion * params.brainstormQuestions
      : params.phaseActive[ph];
  }
  const questions = input.phases.includes('brainstorm') ? params.brainstormQuestions : 0;

import { estimate } from './estimate.mjs';
import { validateInput } from './lib/validate.mjs';

const MIN_SAVING = 5;
const MAX_TIPS = 3;

function ancestors(tasks) {
  const deps = new Map(tasks.map(t => [t.id, t.depends_on ?? []]));
  const result = new Map();
  for (const t of tasks) {
    const seen = new Set();
    const stack = [...deps.get(t.id)];
    while (stack.length) {
      const d = stack.pop();
      if (seen.has(d)) continue;
      seen.add(d);
      stack.push(...(deps.get(d) ?? []));
    }
    result.set(t.id, seen);
  }
  return result;
}

export function independentPairs(tasks) {
  const anc = ancestors(tasks);
  const pairs = [];
  for (let i = 0; i < tasks.length; i++) {
    for (let j = i + 1; j < tasks.length; j++) {
      const a = tasks[i];
      const b = tasks[j];
      const fa = a.files ?? [];
      const fb = b.files ?? [];
      if (!fa.length || !fb.length) continue;
      if (anc.get(a.id).has(b.id) || anc.get(b.id).has(a.id)) continue;
      if (fa.some(f => fb.includes(f))) continue;
      pairs.push([a.id, b.id]);
    }
  }
  return pairs;
}

export function speedups(rawInput, params, base = estimate(rawInput, params)) {
  const input = validateInput(rawInput);
  const saved = options => base.wall - estimate(input, params, options).wall;
  const tips = [];

  let best = null;
  for (const [a, b] of independentPairs(input.tasks)) {
    const s = saved({ parallel: [[a], [b]] });
    if (!best || s > best.saved) best = { a, b, saved: s };
  }
  if (best) {
    tips.push({
      kind: 'parallel',
      saved: best.saved,
      text: `Tasks ${best.a} & ${best.b} are independent → run them in parallel, each in its own git worktree`,
    });
  }

  const phaseGates = base.questions + base.approvals;
  if (phaseGates > 1) {
    tips.push({
      kind: 'frontload',
      saved: saved({ frontLoad: true }),
      text: `${phaseGates} questions/approvals before the build → answer them in one go`,
    });
  }

  if (base.manualTests > 1) {
    tips.push({
      kind: 'batch-tests',
      saved: saved({ batchTests: true }),
      text: `${base.manualTests} manual tests → batch them into one test pass at the end`,
    });
  }

  const out = tips
    .filter(t => t.saved >= MIN_SAVING)
    .sort((x, y) => y.saved - x.saved)
    .slice(0, MAX_TIPS);

  const mech = input.tasks.filter(t => t.kind === 'mechanical').map(t => t.id);
  if (mech.length && out.length < MAX_TIPS) {
    const many = mech.length > 1;
    out.push({
      kind: 'cheaper-model',
      saved: 0,
      text: `Task${many ? 's' : ''} ${mech.join(', ')} ${many ? 'are' : 'is'} mechanical → use a cheaper model (lower cost, ~same time)`,
    });
  }
  return out;
}

export function formatDuration(minutes) {
  const m = Math.max(0, Math.round(minutes));
  if (m < 60) return `${m} min`;
  const h = Math.floor(m / 60);
  const r = m % 60;
  return r ? `${h}h ${String(r).padStart(2, '0')}m` : `${h}h`;
}

const plural = (n, word) => `${n} ${word}${n === 1 ? '' : 's'}`;

export function formatReport(est, tips, status, warning = null) {
  const parts = [];
  if (est.questions) parts.push(plural(est.questions, 'question'));
  if (est.approvals) parts.push(plural(est.approvals, 'approval'));
  if (est.manualTests) parts.push(plural(est.manualTests, 'manual test'));

  const lines = [
    `Total: ${formatDuration(est.low)} – ${formatDuration(est.high)} (wall clock)`,
    `  Agents working:   ~${formatDuration(est.agentMinutes)}   (you can walk away)`,
    `  You needed:       ~${formatDuration(est.userActiveMinutes)}   (${parts.length ? parts.join(', ') : 'no checkpoints'})`,
    `  Waiting on you:   ~${formatDuration(est.replyDelayMinutes)}   (Claude idle until you reply)`,
    `  Biggest risk:     Task ${est.risk.id} — ${est.risk.title} (+${formatDuration(est.risk.extra)} if it goes badly)`,
  ];

  if (tips.length) {
    const total = tips.reduce((s, t) => s + t.saved, 0);
    lines.push('', total >= 1 ? `Speed-ups (−${formatDuration(total)}):` : 'Speed-ups:');
    for (const t of tips) lines.push(`  • ${t.text}${t.saved >= 1 ? ` (−${formatDuration(t.saved)})` : ''}`);
  }

  lines.push('', `Calibration: ${status}`);
  if (warning) lines.push(`Warning: ${warning}`);
  return lines.join('\n');
}

const TASK_HEADING = /^###\s+Task\s+(\d+):\s*(.+?)\s*$/;
const ANY_HEADING = /^#{1,3}\s/;
const FILE_LINE = /^\s*-\s*(?:Create|Modify|Test|Delete):\s*`([^`]+)`/i;
const CHECKBOX = /^\s*-\s\[( |x|X)\]/;

export function parsePlan(text) {
  const lines = text.split(/\r?\n/);
  const workflow = /subagent-driven-development/.test(lines.slice(0, 20).join('\n')) ? 'subagent-driven' : 'inline';
  const tasks = [];
  let cur = null;
  let inFiles = false;
  let inFence = false;
  let totalSteps = 0;
  let checkedSteps = 0;

  for (const line of lines) {
    if (/^\s*```/.test(line)) {
      inFence = !inFence;
      continue;
    }
    if (inFence) continue;

    const h = line.match(TASK_HEADING);
    if (h) {
      cur = { id: Number(h[1]), title: h[2], files: [], steps: 0, done: 0 };
      tasks.push(cur);
      inFiles = false;
      continue;
    }
    if (ANY_HEADING.test(line)) {
      cur = null;
      inFiles = false;
      continue;
    }
    if (!cur) continue;

    if (/^\*\*Files:\*\*/.test(line.trim())) {
      inFiles = true;
      continue;
    }
    if (inFiles) {
      const f = line.match(FILE_LINE);
      if (f) {
        cur.files.push(f[1].replace(/:\d+(?:-\d+)?$/, ''));
        continue;
      }
      if (line.trim() !== '') inFiles = false;
    }

    const c = line.match(CHECKBOX);
    if (c) {
      cur.steps++;
      totalSteps++;
      if (c[1] !== ' ') {
        cur.done++;
        checkedSteps++;
      }
    }
  }

  return { workflow, tasks, totalSteps, checkedSteps, allChecked: totalSteps > 0 && totalSteps === checkedSteps };
}

    if (!text && !c.some(b => b?.type === 'image')) return false;
  } else {
    return false;
  }
  return !NOT_HUMAN.test(text);
}

export function readEntries(file) {
  let text;
  try {
    text = readFileSync(file, 'utf8');
  } catch {
    return [];
  }
  const out = [];
  for (const line of text.split('\n')) {
    if (!line.trim()) continue;
    let e;
    try {
      e = JSON.parse(line);
    } catch {
      continue;
    }
    if ((e.type !== 'user' && e.type !== 'assistant') || !e.timestamp) continue;
    const ts = Date.parse(e.timestamp);
    if (Number.isNaN(ts)) continue;
    out.push({ ts, human: isHumanPrompt(e) });
  }
  return out;
}

export function transcriptFiles(projectDir) {
  const out = { main: [], subagents: [] };
  if (!existsSync(projectDir)) return out;
  for (const entry of readdirSync(projectDir, { withFileTypes: true })) {
    const full = join(projectDir, entry.name);
    if (entry.isFile() && entry.name.endsWith('.jsonl')) {
      out.main.push(full);
    } else if (entry.isDirectory()) {
      const sub = join(full, 'subagents');
      if (!existsSync(sub)) continue;
      for (const f of readdirSync(sub)) if (f.endsWith('.jsonl')) out.subagents.push(join(sub, f));
    }
  }
  return out;
}

const touchedSince = (file, startMs) => statSync(file).mtimeMs >= startMs;

export function measureWindow(projectDir, startMs, endMs, breakMinutes) {
  const { main, subagents } = transcriptFiles(projectDir);
  const entries = [];
  for (const f of [...main, ...subagents]) {
    if (!touchedSince(f, startMs)) continue;
    for (const e of readEntries(f)) if (e.ts >= startMs && e.ts <= endMs) entries.push(e);
  }
  entries.sort((a, b) => a.ts - b.ts);

  const breakMs = breakMinutes * 60000;
  let agentMinutes = 0;
  let userMinutes = 0;
  let humanGaps = 0;
  for (let i = 1; i < entries.length; i++) {
    const gap = entries[i].ts - entries[i - 1].ts;
    if (gap > breakMs) continue;
    if (entries[i].human) {
      userMinutes += gap / 60000;
      humanGaps++;
    } else {
      agentMinutes += gap / 60000;
    }
  }
  return { agentMinutes, userMinutes, humanGaps, entries: entries.length };
}

export function subagentRuns(projectDir, startMs, endMs) {
  const runs = [];
  for (const f of transcriptFiles(projectDir).subagents) {
    if (!touchedSince(f, startMs)) continue;
    const es = readEntries(f).filter(e => e.ts >= startMs && e.ts <= endMs);
    if (es.length < 2) continue;
    let lo = Infinity;
    let hi = -Infinity;
    for (const e of es) {
      if (e.ts < lo) lo = e.ts;
      if (e.ts > hi) hi = e.ts;
    }
    let description = '';
    try {
      description = JSON.parse(readFileSync(f.replace(/\.jsonl$/, '.meta.json'), 'utf8')).description ?? '';
    } catch {
      description = '';
    }
    runs.push({ description, minutes: (hi - lo) / 60000 });
  }
  return runs;
}

import { existsSync, readFileSync, statSync } from 'node:fs';
import { join } from 'node:path';
import { parsePlan } from './lib/plan.mjs';
import { projectsRoot, encodeProjectPath, measureWindow, subagentRuns } from './lib/transcripts.mjs';

export function nudge(old, ratio, params) {
  const [lo, hi] = params.clamp;
  const r = Math.min(hi, Math.max(lo, ratio));
  return old + params.learningRate * (r * old - old);
}

export function findDoneAt(record, now, params) {
  if (record.doneAt) return { status: 'done', doneAt: Date.parse(record.doneAt) };
  if (record.planPath && existsSync(record.planPath)) {
    const plan = parsePlan(readFileSync(record.planPath, 'utf8'));
    if (plan.allChecked) return { status: 'done', doneAt: statSync(record.planPath).mtimeMs };
  }
  if (now - Date.parse(record.startedAt) > params.expiryDays * 86400000) return { status: 'expired' };
  return { status: 'open' };
}

export function matchTasks(perTask, runs) {
  const out = [];
  for (const t of perTask) {
    const byNumber = new RegExp(`\\btask\\s*#?${t.id}\\b`, 'i');
    const title = t.title.toLowerCase();
    const hits = runs.filter(r => byNumber.test(r.description) || r.description.toLowerCase().includes(title));
    if (hits.length) out.push({ ...t, actual: hits.reduce((s, r) => s + r.minutes, 0) });
  }
  return out;
}

export function applyLearning(calibration, record, actual, matched, params) {
  const cal = structuredClone(calibration);
  if (matched.length) {
    for (const t of matched) {
      const key = `${t.kind}.${t.size}`;
      cal.cells[key] = nudge(cal.cells[key] ?? 1, t.actual / t.agent, params);
    }
  } else if (record.predicted.agentMinutes > 0) {
    const w = record.workflow;
    cal.workflow[w] = nudge(cal.workflow[w] ?? 1, actual.agentMinutes / record.predicted.agentMinutes, params);
  }
  if (actual.humanGaps > 0) {
    const old = cal.replyDelay ?? params.replyDelay;
    cal.replyDelay = nudge(old, actual.userMinutes / actual.humanGaps / old, params);
  }
  cal.confirmed += 1;
  return cal;
}

export function runMeasure({ dir, defaults, now = Date.now(), root = projectsRoot() }) {
  const records = readRecords(dir);
  let { calibration } = loadCalibration(dir);
  let changed = false;

  for (const r of records) {
    if (r.status !== 'open') continue;
    const d = findDoneAt(r, now, defaults);
    if (d.status === 'open') continue;
    changed = true;

    if (d.status === 'expired') {
      r.status = 'unconfirmed';
      r.reason = `no done signal within ${defaults.expiryDays} days`;
      continue;
    }

    const projectDir = join(root, encodeProjectPath(r.project));
    const startMs = Date.parse(r.startedAt);
    r.doneAt = new Date(d.doneAt).toISOString();
    const actual = measureWindow(projectDir, startMs, d.doneAt, defaults.breakMinutes);
    if (actual.entries < 2) {
      r.status = 'unconfirmed';
      r.reason = 'no transcript activity found';
      logError(dir, `${r.id}: no transcript activity in ${projectDir}`);
      continue;
    }

    const matched = r.mode === 'precise'
      ? matchTasks(r.predicted.perTask, subagentRuns(projectDir, startMs, d.doneAt))
      : [];
    r.status = 'confirmed';
    r.actual = {
      agentMinutes: actual.agentMinutes,
      userMinutes: actual.userMinutes,
      wall: actual.agentMinutes + actual.userMinutes,
      matchedTasks: matched.length,
    };
    calibration = applyLearning(calibration, r, actual, matched, defaults);
  }

  if (changed) {
    writeRecords(dir, records);
    saveCalibration(dir, calibration);
  }
  return { records, calibration };
}

      } catch (e) {
        err(e.message);
        return 2;
      }
      const tips = speedups(input, params, est);
      const planPath = option(args, '--plan');
      const record = {
        id: newId(now),
        project,
        startedAt: new Date(now).toISOString(),
        mode: planPath ? 'precise' : 'rough',
        planPath: planPath ? resolve(project, planPath) : null,
        workflow: est.workflow,
        input,
        predicted: {
          agentMinutes: est.agentMinutes,
          userMinutes: est.userMinutes,
          wall: est.wall,
          low: est.low,
          high: est.high,
          perTask: est.tasks.map(({ id, title, kind, size, agent }) => ({ id, title, kind, size, agent })),
        },
        status: 'open',
      };
      appendRecord(dir, record);
      out(formatReport(est, tips, calibrationStatus(defaults, calibration), warning));
      out('');
      out(planPath
        ? `Estimate ${record.id} saved. I'll notice when every plan checkbox is ticked, or run /agent-eta:eta-done.`
        : `Estimate ${record.id} saved. Run /agent-eta:eta-done when this work is finished so the estimate can learn.`);
      return 0;
    }

    case 'done': {
      const records = readRecords(dir);
      const target = pos[0]
        ? records.find(r => r.id === pos[0] && r.status === 'open')
        : [...records].reverse().find(r => r.status === 'open' && r.project === project);
      if (!target) {
        out('No open estimate for this project.');
        return 1;
      }
      target.doneAt = new Date(now).toISOString();
      writeRecords(dir, records);
      const result = safeMeasure(ctx);
      const r = result?.records.find(x => x.id === target.id);
      if (r?.status === 'confirmed') {
        out(`Estimate ${r.id} confirmed. Predicted ~${formatDuration(r.predicted.wall)}, actual ~${formatDuration(r.actual.wall)} (${percent(r.actual.wall, r.predicted.wall)}).`);
      } else {
        out(`Estimate ${target.id} could not be measured (${r?.reason ?? 'measurement failed'}). It won't be used for learning.`);
      }
      return 0;
    }

    case 'stats': {
      const result = safeMeasure(ctx) ?? { records: readRecords(dir), calibration: loadCalibration(dir).calibration };
      const confirmed = result.records.filter(r => r.status === 'confirmed').slice(-10);
      const lines = [];
      if (!confirmed.length) {
        lines.push('No confirmed estimates yet. Run /agent-eta:eta-done when estimated work is finished.');
      } else {
        lines.push(`agent-eta — last ${confirmed.length} confirmed estimate${confirmed.length === 1 ? '' : 's'}`);
        for (const r of confirmed) {
          lines.push(`  ${r.id}  ${r.startedAt.slice(0, 10)}  predicted ${formatDuration(r.predicted.wall)}  actual ${formatDuration(r.actual.wall)}  ${percent(r.actual.wall, r.predicted.wall)}`);
        }
        const avg = confirmed.reduce((s, r) => s + (r.actual.wall - r.predicted.wall) / r.predicted.wall, 0) / confirmed.length;
        const sign = Math.round(avg * 100);
        lines.push(`Average error: ${sign > 0 ? '+' : ''}${sign}% (${sign > 0 ? 'estimates run low' : sign < 0 ? 'estimates run high' : 'on target'})`);
      }
      const open = result.records.filter(r => r.status === 'open').length;
      const unconfirmed = result.records.filter(r => r.status === 'unconfirmed').length;
      lines.push(`Open: ${open} · Unconfirmed: ${unconfirmed}`);
      const cal = result.calibration;
      const corr = [
        ...Object.entries(cal.cells).map(([k, v]) => `${k} ×${v.toFixed(2)}`),
        ...Object.entries(cal.workflow).map(([k, v]) => `workflow ${k} ×${v.toFixed(2)}`),
        ...(cal.replyDelay != null ? [`reply delay ${cal.replyDelay.toFixed(1)} min`] : []),
      ];
      if (corr.length) lines.push(`Corrections: ${corr.join(', ')}`);
      lines.push(`Calibration: ${calibrationStatus(defaults, cal)}`);
      out(lines.join('\n'));
      return 0;
    }

    case 'measure': {
      const result = safeMeasure(ctx);
      out(result ? `Measured. ${result.records.filter(r => r.status === 'confirmed').length} confirmed estimates.` : 'Measure failed; see errors.log.');
      return result ? 0 : 1;
    }

    default:
      out(USAGE);
      return 1;
  }
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  process.exitCode = main(process.argv.slice(2));
}

```js
// Dev-only. Reads the author's local Claude Code transcripts and derives starting numbers.
// Only aggregate numbers are written; no transcript text, file names or project names.
// Usage: node tools/seed-defaults.mjs          (print only)
//        node tools/seed-defaults.mjs --write  (also update data/defaults.json)
import { readdirSync, existsSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { projectsRoot, readEntries } from '../scripts/lib/transcripts.mjs';
import { loadDefaults, DEFAULTS_PATH } from '../scripts/lib/config.mjs';

const MIN_RUN = 0.5;
const MAX_RUN = 120;
const MAX_REPLY = 30;
const KIND_FACTOR = { mechanical: 0.6, integration: 1.0, judgment: 1.6 };

const round1 = x => Math.round(x * 10) / 10;
function pct(xs, p) {
  const s = [...xs].sort((a, b) => a - b);
  return s[Math.min(s.length - 1, Math.floor(p * (s.length - 1)))];
}

const root = projectsRoot();
const runs = [];
const replies = [];

for (const proj of readdirSync(root, { withFileTypes: true })) {
  if (!proj.isDirectory()) continue;
  const pdir = join(root, proj.name);
  for (const entry of readdirSync(pdir, { withFileTypes: true })) {
    if (entry.isFile() && entry.name.endsWith('.jsonl')) {
      const es = readEntries(join(pdir, entry.name)).sort((a, b) => a.ts - b.ts);
      for (let i = 1; i < es.length; i++) {
        if (!es[i].human) continue;
        const gap = (es[i].ts - es[i - 1].ts) / 60000;
        if (gap > 0 && gap <= MAX_REPLY) replies.push(gap);
      }
    } else if (entry.isDirectory()) {
      const sub = join(pdir, entry.name, 'subagents');
      if (!existsSync(sub)) continue;
      for (const f of readdirSync(sub)) {
        if (!f.endsWith('.jsonl')) continue;
        const es = readEntries(join(sub, f));
        if (es.length < 2) continue;
        let lo = Infinity;
        let hi = -Infinity;
        for (const e of es) {
          if (e.ts < lo) lo = e.ts;
          if (e.ts > hi) hi = e.ts;
        }
        const minutes = (hi - lo) / 60000;
        if (minutes >= MIN_RUN && minutes <= MAX_RUN) runs.push(minutes);
      }
    }
  }
}

if (runs.length < 30 || replies.length < 30) {
  console.error(`Not enough data (runs=${runs.length}, replies=${replies.length}); need 30 of each.`);
  process.exit(1);
}

const [p25, p50, p75] = [0.25, 0.5, 0.75].map(p => pct(runs, p));
const base = {};
for (const [kind, f] of Object.entries(KIND_FACTOR)) {
  base[kind] = { S: round1(p25 * f), M: round1(p50 * f), L: round1(p75 * f) };
}
const replyDelay = round1(pct(replies, 0.5));

console.log(`Subagent runs: ${runs.length}  p25=${round1(p25)}  p50=${round1(p50)}  p75=${round1(p75)} min`);
console.log(`Human replies: ${replies.length}  median=${replyDelay} min`);
console.log('base =', JSON.stringify(base));

if (process.argv.includes('--write')) {
  const d = loadDefaults();
  d.base = base;
  d.replyDelay = replyDelay;
  writeFileSync(DEFAULTS_PATH, `${JSON.stringify(d, null, 2)}\n`);
  console.log(`Wrote ${DEFAULTS_PATH}`);
}
```

Ask Claude "how long will this take?" and you get a human developer's estimate. With Claude Code and subagents, typing is nearly free. The real time goes to waiting on you, verification, tasks that must run in order, and rework.

## How it learns

Each estimate is saved locally. It separates agent time from your reply time and ignores breaks over 30 minutes. Each result nudges your personal numbers 30% of the way toward reality, capped at ×0.5–×2 per update. Everything stays on your machine. Estimates and calibration live in the plugin data folder (or `~/.claude/agent-eta/`). Nothing is uploaded.

// Dev-only. Runs the eta skill N times through `claude -p` on the same description
// and checks every run's range midpoint is within ±20% of the mean.
// Usage: node tools/consistency.mjs "<feature description>" [runs=3]
import { spawnSync } from 'node:child_process';
import { mkdtempSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join, resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const description = process.argv[2];
const runs = Number(process.argv[3] ?? 3);
if (!description) {
  console.error('Usage: node tools/consistency.mjs "<feature description>" [runs=3]');
  process.exit(1);
}

const pluginDir = resolve(dirname(fileURLToPath(import.meta.url)), '..');
const home = mkdtempSync(join(tmpdir(), 'agent-eta-consistency-'));

function toMinutes(s) {
  const h = s.match(/(\d+)h/);
  const m = s.match(/(\d+)\s*(?:m\b|min)/);
  return (h ? Number(h[1]) * 60 : 0) + (m ? Number(m[1]) : 0);
}

const mids = [];
for (let i = 1; i <= runs; i++) {
  const r = spawnSync('claude', [
    '-p', `Use the agent-eta:eta skill to estimate: ${description}`,
    '--plugin-dir', pluginDir,
    '--allowedTools', 'Bash', 'Read', 'Glob', 'Grep', 'Write', 'Agent', 'Skill',
  ], { encoding: 'utf8', env: { ...process.env, AGENT_ETA_HOME: home }, timeout: 600000 });
  if (r.error) {
    console.error(`Run ${i} failed to start: ${r.error.message}`);
    process.exit(1);
  }
  const m = (r.stdout ?? '').match(/Total:\s*(.+?)\s*–\s*(.+?)\s*\(wall clock\)/);
  if (!m) {
    console.error(`Run ${i}: no "Total:" line found. Output:\n${r.stdout}\n${r.stderr}`);
    process.exit(1);
  }
  const mid = (toMinutes(m[1]) + toMinutes(m[2])) / 2;
  mids.push(mid);
  console.log(`Run ${i}: ${m[1]} – ${m[2]}  (midpoint ${mid} min)`);
}

const mean = mids.reduce((s, x) => s + x, 0) / mids.length;
const worst = Math.max(...mids.map(x => Math.abs(x - mean) / mean));
console.log(`Mean midpoint ${mean.toFixed(1)} min; worst deviation ${(worst * 100).toFixed(0)}%`);
if (worst > 0.2) {
  console.log('FAIL: runs disagree by more than 20%. Tighten the tagging rules in skills/eta/SKILL.md.');
  process.exit(1);
}
console.log('PASS');
```
